# ---------------------------------------------------------------------------
# 视觉（模块 A）
# ---------------------------------------------------------------------------
# ⚠️ 只保留 contrib 这一个 cv2 提供者。
#
# opencv-python / opencv-contrib-python / opencv-python-headless 三者**共享
# 同一个 `cv2` 目录**。同时安装时谁生效取决于安装顺序，而且**卸载其中任何一个
# 都会删掉另外两个的文件**（因为删的是那个共享目录）。表现出来就是
# 「昨天还好好的」这类极难排查的故障 —— 本机就真的踩过一次。
#
# 必须用 contrib：mediapipe 需要它。安装/修复请用
#     pip install --force-reinstall --no-deps opencv-contrib-python==4.13.0.92
# 单独 pip uninstall 某一个再装回来是**修不好**的。
opencv-contrib-python==4.13.0.92

# 人脸关键点。锁 0.10.35 而不是 >=1.0：
#   - 模块 A 用的是 **Tasks API**（mediapipe.tasks.python），0.10.x 就有；
#   - pip 会把 `mediapipe>=1.0` 解析到 1.0.1，那是跨大版本的另一套东西，
#     wheel 标签(py3-none-win_amd64)都与已装的 0.10.35 原生包不同形态。
#   - 本机 0.10.35 已实测可导入，Python 3.14 的兼容问题实际不存在。
mediapipe==0.10.35

numpy>=1.26

# ---------------------------------------------------------------------------
# 界面（模块 C）
# ---------------------------------------------------------------------------
# 写 5.15.11 是因为官方 wheel 只出到 cp38-abi3，而 abi3 向前兼容，能装在 3.14 上。
# ⚠️ 验证必须用下面这条，`import PyQt5` 在**残缺安装**上也会成功（假阳性）：
#     python -c "from PyQt5.QtWidgets import QApplication; print('OK')"
PyQt5==5.15.11

# ---------------------------------------------------------------------------
# 测试
# ---------------------------------------------------------------------------
# 三个模块的测试都在仓库根一条命令跑完：pytest
pytest==9.1.1

# ---------------------------------------------------------------------------
# Django 脚手架
# ---------------------------------------------------------------------------

Django>=5.2
